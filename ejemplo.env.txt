# Copiar a .env y completar. El .env NO se sube al repositorio.

# ------------------------------------------------------------------ llaves

# Llave maestra con la que se cifran certificados y CAF en la base de datos.
# Si se pierde, no se pueden leer los certificados guardados: respáldala aparte,
# fuera de este servidor. Un respaldo de la base sin esta llave no sirve de nada.
# Generar con:  openssl rand -base64 48
DTE_LLAVE_MAESTRA=

# Llave de administración: da de alta emisores y carga certificados y CAF.
# Sin ella esas rutas quedan cerradas y el servicio arranca igual, solo avisando
# por consola. Tiene que tener al menos 24 caracteres.
# Generar con:  openssl rand -base64 36
DTE_LLAVE_ADMIN=

# ------------------------------------------------------------ base de datos

# La clave va también en POSTGRES_PASSWORD del compose: son la misma.
# Generar con:  openssl rand -hex 24
POSTGRES_PASSWORD=
DATABASE_URL=postgres://dte:LA_MISMA_CLAVE@db:5432/dte

# ----------------------------------------------------------------- servicio

# Ambiente por defecto de los emisores nuevos: certificacion | produccion
DTE_AMBIENTE=certificacion

# Cada cuánto se les pregunta al SII por los documentos enviados, en milisegundos.
DTE_SEGUIMIENTO_MS=60000

PUERTO=3000

# ---------------------------------------------------------------------- TLS

# El dominio con el que se publica el servicio. Caddy saca el certificado solo,
# así que este nombre tiene que apuntar por DNS a la IP del servidor ANTES de
# levantar el compose.
DOMINIO=dte.ejemplo.cl

# Correo para los avisos de vencimiento del certificado TLS (Let's Encrypt).
CORREO_TLS=
